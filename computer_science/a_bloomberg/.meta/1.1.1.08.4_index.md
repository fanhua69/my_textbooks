# 1.1.1.08.4 CPP20_SnapShot_Example

**介绍**: CPP20_SnapShot_Example

```cpp
#include <atomic>
#include <chrono>
#include <iostream>
#include <memory>
#include <string>
#include <thread>
#include <unordered_map>
#include <unordered_set>
#include <vector>

// 不可变权限快照：一旦构造完成，只读不改
struct PermissionSnapshot {
    uint64_t version = 0;
    // user_id -> 拥有的权限集合
    std::unordered_map<std::string, std::unordered_set<std::string>> user_perms;

    bool has_permission(const std::string& user, const std::string& perm) const {
        auto it = user_perms.find(user);
        if (it == user_perms.end()) {
            return false;
        }
        return it->second.count(perm) > 0;
    }
};

struct PermissionEvent {
    uint64_t event_id;
    std::string user;
    std::string perm;
    bool grant; // true = 授予, false = 撤销
};

class PermissionStore {
public:
    PermissionStore() {
        auto init_snap = std::make_shared<const PermissionSnapshot>();
        current_snapshot_.store(init_snap, std::memory_order_release);
    }

    // 在线请求线程调用：无锁获取当前快照副本，随后安全只读查询
    std::shared_ptr<const PermissionSnapshot> acquire_snapshot() const {
        return current_snapshot_.load(std::memory_order_acquire);
    }

    // 异步同步线程 / Consumer 调用：基于旧快照构建新快照并原子切换
    void apply_events(const std::vector<PermissionEvent>& events) {
        if (events.empty()) {
            return;
        }

        // 1. 拿到当前最新快照作为构建基线（单 Consumer 写入时无需锁；多写者需加写锁序列化构建）
        std::shared_ptr<const PermissionSnapshot> old_snap =
            current_snapshot_.load(std::memory_order_acquire);

        // 2. 拷贝一份副本进行离场修改（Copy-on-Write 思路）
        auto next_snap = std::make_shared<PermissionSnapshot>(*old_snap);

        for (const auto& ev : events) {
            // 防止乱序/重复事件回退版本
            if (ev.event_id <= next_snap->version) {
                continue;
            }
            auto& perms = next_snap->user_perms[ev.user];
            if (ev.grant) {
                perms.insert(ev.perm);
            } else {
                perms.erase(ev.perm);
                if (perms.empty()) {
                    next_snap->user_perms.erase(ev.user);
                }
            }
            next_snap->version = ev.event_id;
        }

        // 3. 转换为 const 快照并原子发布（release 保证 next_snap 内容对 acquire 读者可见）
        std::shared_ptr<const PermissionSnapshot> published = std::move(next_snap);
        current_snapshot_.store(std::move(published), std::memory_order_release);
    }

private:
    // C++20 支持对 std::shared_ptr 使用 std::atomic 特化
    std::atomic<std::shared_ptr<const PermissionSnapshot>> current_snapshot_;
};

int main() {
    PermissionStore store;
    std::atomic<bool> stop{false};

    // 模拟多个在线请求线程持续查询权限
    std::vector<std::thread> readers;
    for (int t = 0; t < 3; ++t) {
        readers.emplace_back([&, t]() {
            while (!stop.load(std::memory_order_relaxed)) {
                auto snap = store.acquire_snapshot();
                bool allowed = snap->has_permission("alice", "order.refund");
                if (allowed) {
                    // 仅打印少量命中日志，避免刷屏
                    static std::atomic<int> logged{0};
                    if (logged.fetch_add(1, std::memory_order_relaxed) < 5) {
                        std::cout << "[Reader " << t << "] v" << snap->version
                                  << " alice has order.refund = true\n";
                    }
                }
            }
        });
    }

    // 模拟异步同步线程 / Consumer 陆续消费事件日志并发布新快照
    std::thread consumer([&]() {
        std::this_thread::sleep_for(std::chrono::milliseconds(10));
        store.apply_events({
            {1, "alice", "order.read", true},
            {2, "alice", "order.refund", true},
        });
        std::cout << "[Consumer] Published snapshot v2\n";

        std::this_thread::sleep_for(std::chrono::milliseconds(20));
        store.apply_events({
            {3, "alice", "order.refund", false},
        });
        std::cout << "[Consumer] Published snapshot v3 (revoked order.refund)\n";
    });

    consumer.join();
    std::this_thread::sleep_for(std::chrono::milliseconds(20));
    stop.store(true, std::memory_order_relaxed);

    for (auto& th : readers) {
        th.join();
    }

    auto final_snap = store.acquire_snapshot();
    std::cout << "[Final] version=" << final_snap->version
              << ", alice order.refund="
              << (final_snap->has_permission("alice", "order.refund") ? "true" : "false")
              << "\n";
    return 0;
}
```
